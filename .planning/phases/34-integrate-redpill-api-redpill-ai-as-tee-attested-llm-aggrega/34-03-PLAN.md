   pub async fn check_model_routable(
       backend: &BackendConfig,
       model: &str,
   ) -> Result<(), crate::attestation::redpill::RedpillError> {
       use crate::attestation::redpill::RedpillError;

       let url = model_list_url(backend)
           .map_err(|e| RedpillError::Network(format!("model_list_url: {e:?}")))?;

       let client = build_http_client(Duration::from_secs(30))
           .map_err(|e| RedpillError::Network(format!("client: {e:?}")))?;

       let resp = client.get(&url).send().await
           .map_err(|e| RedpillError::Network(e.to_string()))?;
       let body: serde_json::Value = resp.json().await
           .map_err(|e| RedpillError::Network(format!("parse models: {e}")))?;

       // Models response shape: { data: [{ id: "...", providers: ["phala"|"tinfoil"|"chutes"|...] }, ...] }
       let entries = body.get("data").and_then(|d| d.as_array());
       let Some(entries) = entries else { return Ok(()); };

       for entry in entries {
           if entry.get("id").and_then(|i| i.as_str()) == Some(model) {
               if let Some(providers) = entry.get("providers").and_then(|p| p.as_array()) {
                   if providers.iter().any(|p| p.as_str() == Some("tinfoil")) {
                       return Err(RedpillError::TinfoilUnsupported);
                   }
               }
               return Ok(());
           }
       }
       Ok(()) // Unknown model — let the attestation fetch surface a more precise error
   }
