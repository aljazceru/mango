    ```rust
    //! Phase 35 — Tool Discovery sub-screen for the Desktop (iced) UI.

    use iced::{
        widget::{button, column, container, row, scrollable, text, toggler, Space},
        Alignment, Length, Padding,
    };

    use crate::theme::ViewColors;
    use crate::Message;

    use mango_core::{AppState, ContextvmDiscoveryState, DiscoverableTool, Screen};

    pub fn view<'a>(state: &'a crate::AppModel, vc: &ViewColors)
        -> iced::Element<'a, Message> {

        let header = row![
            button(text("Back").size(13).color(vc.text))
                .on_press(Message::PopScreen)
                .padding(Padding::from([6u16, 12])),
            Space::new().width(Length::Fill),
            text("Discover Tools").size(17).color(vc.text),
            Space::new().width(Length::Fill),
            // Refresh button — disabled while Loading.
            {
                let mut b = button(text("Refresh").size(13).color(vc.text))
                    .padding(Padding::from([6u16, 12]));
                if !matches!(state.app_state.contextvm_discovery_state,
                             ContextvmDiscoveryState::Loading) {
                    b = b.on_press(Message::ContextvmRetryClicked);
                }
                b
            },
        ]
        .align_y(Alignment::Center)
        .spacing(8)
        .padding(Padding::from([8u16, 16]));

        let body: iced::Element<Message> = match &state.app_state.contextvm_discovery_state {
            ContextvmDiscoveryState::Idle | ContextvmDiscoveryState::Loading => {
                centered_message(vc, "Searching Nostr relays…", None, None, false)
            }
            ContextvmDiscoveryState::Error { .. } => {
                centered_message(
                    vc,
                    "Couldn’t reach relays",
                    Some("Check your connection and try again."),
                    Some("Try again"),
                    true, // destructive heading color
                )
            }
            ContextvmDiscoveryState::Loaded => {
                if state.app_state.contextvm_tools.is_empty() {
                    centered_message(
                        vc,
                        "No tools found",
                        Some("Tools advertised on Nostr will appear here."),
                        Some("Try again"),
                        false,
                    )
                } else {
                    tool_list(state, vc)
                }
            }
        };

        column![header, body]
            .spacing(8)
            .into()
    }

    fn centered_message<'a>(
        vc: &ViewColors,
        headline: &'a str,
        body_text: Option<&'a str>,
        button_label: Option<&'a str>,
        destructive: bool,
    ) -> iced::Element<'a, Message> {
        let headline_color = if destructive { vc.destructive } else { vc.text };
        let mut col = column![
            text(headline).size(16).color(headline_color),
        ].spacing(12).align_x(Alignment::Center);
        if let Some(b) = body_text {
            col = col.push(text(b).size(14).color(vc.muted));
        }
        if let Some(label) = button_label {
            col = col.push(
                button(text(label).size(13).color(vc.text))
                    .on_press(Message::ContextvmRetryClicked)
                    .padding(Padding::from([6u16, 16])),
            );
        }
        container(col)
            .center_x(Length::Fill)
            .center_y(Length::Fill)
            .padding(48)
            .into()
    }

    fn tool_list<'a>(
        state: &'a crate::AppModel,
        vc: &ViewColors,
    ) -> iced::Element<'a, Message> {
        let rows: Vec<iced::Element<Message>> = state
            .app_state
            .contextvm_tools
            .iter()
            .map(|t| tool_row(t, vc).into())
            .collect();
        scrollable(
            column(rows)
                .spacing(8)
                .padding(Padding::from([8u16, 16])),
        )
        .into()
    }

    fn tool_row<'a>(
        tool: &'a DiscoverableTool,
        vc: &'a ViewColors,
    ) -> iced::Element<'a, Message> {
        let provider_label = tool
            .provider_display_name
            .clone()
            .unwrap_or_else(|| {
                let prefix: String = tool.provider_pubkey.chars().take(8).collect();
                format!("{}…", prefix)
            });
        let id = tool.id.clone();
        container(
            row![
                column![
                    text(tool.name.clone()).size(14).color(vc.text),
                    text(provider_label).size(12).color(vc.muted),
                    text(tool.description.clone()).size(12).color(vc.muted),
                ]
                .spacing(2)
                .width(Length::Fill),
                toggler(tool.enabled)
                    .on_toggle(move |enabled| Message::ContextvmToolToggled {
                        tool_id: id.clone(), enabled,
                    })
                    .size(20),
            ]
            .align_y(Alignment::Center)
            .spacing(8),
        )
        .padding(Padding::from([10u16, 16]))
        .width(Length::Fill)
        .into()
    }
    ```
