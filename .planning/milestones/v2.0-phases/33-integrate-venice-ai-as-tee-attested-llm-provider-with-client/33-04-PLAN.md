   ```rust
   //! Live integration test against api.venice.ai.
   //! Run: VENICE_API_KEY=... cargo test -p mango_core --lib live_venice -- --ignored --nocapture

   use crate::config::{BackendConfig, TeeType};
   use crate::llm::backend::ProviderKind;

   fn skip_if_no_key() -> Option<String> {
       match std::env::var("VENICE_API_KEY") {
           Ok(k) if !k.is_empty() => Some(k),
           _ => {
               eprintln!("VENICE_API_KEY not set; skipping live test");
               None
           }
       }
   }

   #[tokio::test]
   #[ignore = "live integration test against api.venice.ai; requires VENICE_API_KEY"]
   async fn live_attestation_verifies() {
       let Some(api_key) = skip_if_no_key() else { return; };
       let backend = BackendConfig {
           id: "venice-ai".into(),
           base_url: "https://api.venice.ai/api/v1/".into(),
           api_key,
           preferred_model: Some("e2ee-venice-uncensored-24b-p".into()),
           tee_type: TeeType::IntelTdx,
           ..Default::default()
       };
       let policy = crate::attestation::policy::TdxPolicy::default();
       let verified = crate::attestation::venice::ensure_verified_venice_attestation(
           &backend, "e2ee-venice-uncensored-24b-p", &policy,
       ).await.expect("attestation must succeed");
       assert_eq!(verified.signing_pubkey_uncompressed[0], 0x04);
       assert_eq!(verified.submitted_nonce.len(), 32);
       assert_eq!(verified.report_blob.len() >= 48, true);
   }

   #[tokio::test]
   #[ignore = "live integration test; requires VENICE_API_KEY"]
   async fn live_chat_completion_e2ee() {
       let Some(api_key) = skip_if_no_key() else { return; };
       let backend = BackendConfig {
           id: "venice-ai".into(),
           base_url: "https://api.venice.ai/api/v1/".into(),
           api_key,
           preferred_model: Some("e2ee-venice-uncensored-24b-p".into()),
           tee_type: TeeType::IntelTdx,
           ..Default::default()
       };

       use async_openai::types::chat::*;
       let messages = vec![
           ChatCompletionRequestUserMessageArgs::default()
               .content("Reply with the single word: VERIFIED")
               .build().unwrap()
               .into(),
       ];

       let response = crate::llm::venice::create_chat_completion(
           backend,
           "e2ee-venice-uncensored-24b-p".into(),
           messages,
           None,
       ).await.expect("chat completion must round-trip via E2EE");

       let content = response.choices[0].message.content.as_deref().unwrap_or("");
       eprintln!("[live] decrypted reply: {content}");
       assert!(!content.is_empty(), "reply must be non-empty plaintext after E2EE decrypt");
   }
   ```
