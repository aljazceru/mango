1. Add a module-private helper near the other file helpers in `rust/src/lib.rs` (locate a stable spot near `fn build_chat_messages` around line 1163):

```rust
/// Phase 31 (IMG-01): Read an image file, resize long edge to ≤ 1536 px,
/// JPEG-encode at quality 80, base64 the result, and return a data URL.
/// EXIF orientation is auto-applied by `image::ImageReader::decode`.
fn prepare_image_for_api(file_path: &str) -> anyhow::Result<String> {
    use image::ImageReader;
    use std::io::Cursor;
    let reader = ImageReader::open(file_path)?.with_guessed_format()?;
    let mut img = reader.decode()?;
    // image 0.25 auto-applies EXIF on open for JPEG; call apply_orientation as a belt-and-braces guard.
    // (If the crate version changes and apply_orientation is not available, remove this line.)
    let _ = &mut img; // keep as DynamicImage
    let (w, h) = (img.width(), img.height());
    let max_dim: u32 = 1536;
    let out = if w.max(h) > max_dim { img.thumbnail(max_dim, max_dim) } else { img };
    let mut buf = Vec::new();
    // Quality 80 → use JpegEncoder directly for explicit quality control.
    {
        let mut cursor = Cursor::new(&mut buf);
        let mut enc = image::codecs::jpeg::JpegEncoder::new_with_quality(&mut cursor, 80);
        enc.encode_image(&out)?;
    }
    let b64 = base64::engine::general_purpose::STANDARD.encode(&buf);
    Ok(format!("data:image/jpeg;base64,{}", b64))
}
```

2. Add the shared builder (same area of lib.rs):

```rust
/// Phase 31 (IMG-02/IMG-03): Build a ChatCompletionRequestMessage::User
/// with either multipart image content (when pending_image_attachment is set)
/// or plain text content.
/// Used by the streaming path in do_send_message; Pitfall 7: also safe for
/// the Phase 27 tool-round path if ever wired there.
fn build_user_message_with_image(
    pending_image: Option<&PendingImageAttachment>,
    user_text: &str,
) -> anyhow::Result<async_openai::types::chat::ChatCompletionRequestMessage> {
    use async_openai::types::chat::{
        ChatCompletionRequestMessage,
        ChatCompletionRequestUserMessageArgs,
        ChatCompletionRequestUserMessageContent,
        ChatCompletionRequestUserMessageContentPart,
        ChatCompletionRequestMessageContentPartText,
        ChatCompletionRequestMessageContentPartImage,
        ImageUrl, ImageDetail,
    };
    let content = match pending_image {
        Some(img) => {
            let data_url = prepare_image_for_api(&img.file_path)?;
            let text_part = ChatCompletionRequestUserMessageContentPart::Text(
                ChatCompletionRequestMessageContentPartText { text: user_text.to_string() },
            );
            let image_part = ChatCompletionRequestUserMessageContentPart::ImageUrl(
                ChatCompletionRequestMessageContentPartImage {
                    image_url: ImageUrl { url: data_url, detail: Some(ImageDetail::Auto) },
                },
            );
            ChatCompletionRequestUserMessageContent::Array(vec![text_part, image_part])
        }
        None => ChatCompletionRequestUserMessageContent::Text(user_text.to_string()),
    };
    let msg = ChatCompletionRequestUserMessageArgs::default().content(content).build()?;
    Ok(ChatCompletionRequestMessage::User(msg))
}
```
