```rust
Message::AttachFile => {
    return iced::Task::perform(async move {
        let path = rfd::AsyncFileDialog::new()
            .add_filter("Attachable", &["jpg", "jpeg", "png", "txt", "md", "json", "csv", "log", "pdf"])
            .add_filter("Images", &["jpg", "jpeg", "png"])
            .add_filter("Text", &["txt", "md", "json", "csv", "log"])
            .pick_file()
            .await?;
        let path_buf = path.path().to_path_buf();
        Some(path_buf)
    }, Message::AttachFileChosen);
}
// New message variant — add to Message enum:
Message::AttachFileChosen(Some(path)) => {
    let filename = path.file_name().map(|n| n.to_string_lossy().into_owned())
        .unwrap_or_else(|| "attachment".to_string());
    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("").to_ascii_lowercase();
    let is_image = matches!(ext.as_str(), "jpg" | "jpeg" | "png");
    let abs_path = match path.canonicalize() { Ok(p) => p, Err(_) => path.clone() };

    if is_image {
        let mime = if ext == "png" { "image/png" } else { "image/jpeg" };
        manager_clone.dispatch(AppAction::AttachImage {
            filename,
            file_path: abs_path.to_string_lossy().into_owned(),
            mime_type: mime.to_string(),
        });
    } else {
        // Existing text-file path — read bytes and send as UTF-8.
        if let Ok(bytes) = std::fs::read(&abs_path) {
            let size = bytes.len() as u64;
            let content = match String::from_utf8(bytes) {
                Ok(s) => s,
                Err(_) => {
                    // Not UTF-8 — surface as error; do not attach binary-as-text.
                    // Consider dispatching a no-op or set last_error; here we just bail.
                    return iced::Task::none();
                }
            };
            manager_clone.dispatch(AppAction::AttachFile { filename, content, size_bytes: size });
        }
    }
    iced::Task::none()
}
Message::AttachFileChosen(None) => iced::Task::none(),
```
