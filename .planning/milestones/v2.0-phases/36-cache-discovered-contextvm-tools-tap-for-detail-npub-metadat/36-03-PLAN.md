    //! Phase 36 — per-tool detail view (Desktop iced).
    //!
    //! Reached via PushScreen { Screen::ContextvmToolDetail { tool_id } }.
    //! Reads the tool out of state.contextvm_tools by id; renders heading,
    //! ADVERTISED BY, USAGE, SCHEMA expander, and Tool ID row.
    //!
    //! All copy strings are locked verbatim from 36-UI-SPEC.md §Copywriting.

    use iced::widget::{button, column, container, row, scrollable, text, Column, Space};
    use iced::{Alignment, Element, Font, Length, Padding};

    use crate::theme::view_colors;
    use crate::Message;
    // ... import AppState, AppAction, Screen, DiscoverableTool from the cdylib bindings ...

    pub fn view<'a>(
        state: &'a AppState,
        tool_id: &'a str,
        copy_status: Option<&'a str>,
        schema_expanded: bool,
        is_dark: bool,
    ) -> Element<'a, Message> {
        let vc = view_colors(is_dark);

        // ── Header bar (back arrow + title) — locked title "Tool details" ──
        let header = row![
            button(text("←").size(20))
                .on_press(Message::DispatchAction(AppAction::PopScreen))
                .style(button::text),
            text("Tool details").size(17),
        ]
        .spacing(12)
        .align_y(Alignment::Center)
        .padding(Padding::from([12u16, 16]));

        // ── Body ──
        let tool = state.contextvm_tools.iter().find(|t| t.id == tool_id);
        let Some(tool) = tool else {
            return column![
                header,
                container(text("Tool not found").size(14).color(vc.muted))
                    .padding(32),
            ].into();
        };

        let mut col: Column<Message> = Column::new().spacing(24).padding(Padding::from([16u16, 16]));

        // Heading block: name + optional caption + description
        col = col.push(text(&tool.name).size(20));
        if tool.usage_count > 0 {
            if let Some(label) = &tool.last_used_label {
                let caption = if tool.usage_count == 1 {
                    format!("Used 1× — last used {}", label)
                } else {
                    format!("Used {}× — last used {}", tool.usage_count, label)
                };
                col = col.push(text(caption).size(12).color(vc.muted));
            }
        }
        if !tool.description.trim().is_empty() {
            col = col.push(text(&tool.description).size(13));
        }

        // ADVERTISED BY section
        let mut adv = Column::new().spacing(4);
        adv = adv.push(text("ADVERTISED BY").size(11).color(vc.muted));  // locked, uppercase by content
        let provider = tool.provider_display_name.clone().unwrap_or_else(|| "Unnamed provider".into());
        adv = adv.push(text(provider).size(14));
        adv = adv.push(copy_row(&tool.npub, None, tool.npub.clone(), Message::CopyNpub(tool.npub.clone()), &vc));
        adv = adv.push(copy_row(
            &format!("{}…", &tool.provider_pubkey[..tool.provider_pubkey.len().min(8)]),
            Some("Hex:"),
            tool.provider_pubkey.clone(),
            Message::CopyHex(tool.provider_pubkey.clone()),
            &vc,
        ));
        col = col.push(adv);

        // USAGE section
        let mut usage = Column::new().spacing(4);
        usage = usage.push(text("USAGE").size(11).color(vc.muted));
        if tool.usage_count == 0 {
            usage = usage.push(text("Never used").size(13).color(vc.muted));
        } else {
            let times = if tool.usage_count == 1 {
                "Used 1 time".to_string()
            } else {
                format!("Used {} times", tool.usage_count)
            };
            usage = usage.push(text(times).size(13));
            if let Some(label) = &tool.last_used_label {
                usage = usage.push(text(format!("Last used {}", label)).size(13).color(vc.muted));
            }
        }
        col = col.push(usage);

        // SCHEMA expander section
        let mut schema_section = Column::new().spacing(4);
        let schema_header = if !tool.schema_pretty.trim().is_empty() {
            row![
                text("SCHEMA").size(11).color(vc.muted),
                Space::with_width(Length::Fill),
                button(text(if schema_expanded { "▲ Hide" } else { "▼ Show" }).size(11))
                    .on_press(Message::ToggleSchemaExpanded)
                    .style(button::text),
            ].align_y(Alignment::Center)
        } else {
            row![text("SCHEMA").size(11).color(vc.muted)]
        };
        schema_section = schema_section.push(schema_header);
        if tool.schema_pretty.trim().is_empty() {
            schema_section = schema_section.push(text("No schema published").size(13).color(vc.muted));
        } else if schema_expanded {
            schema_section = schema_section.push(
                container(
                    scrollable(text(&tool.schema_pretty).size(12).font(Font::MONOSPACE))
                        .height(Length::Fixed(320.0)),
                )
                .padding(12)
                // bg: vc.card; border: 1px vc.border; corner: 8px — match settings.rs card style
            );
        }
        col = col.push(schema_section);

        // Tool ID row
        col = col.push(copy_row(
            &format!("{}…", &tool.id[..tool.id.len().min(8)]),
            Some("Tool ID:"),
            tool.id.clone(),
            Message::CopyToolId(tool.id.clone()),
            &vc,
        ));

        // Optional copy-confirmation status line
        if let Some(status) = copy_status {
            // Color: vc.success for affirmative ("npub copied" / "Pubkey copied" / "Tool ID copied"),
            // vc.destructive only for the failure-path "Couldn't copy — try again".
            let color = if status.starts_with("Couldn't") { vc.destructive } else { vc.success };
            col = col.push(text(status).size(11).color(color));
        }

        column![header, scrollable(col)].into()
    }

    fn copy_row<'a>(
        display: &'a str,
        prefix: Option<&'a str>,
        full_value: String,
        on_copy: Message,
        vc: &crate::theme::ViewColors,
    ) -> Element<'a, Message> {
        let mut r = row![].spacing(8).align_y(Alignment::Center);
        if let Some(p) = prefix {
            r = r.push(text(p).size(13).color(vc.muted));
        }
        r = r.push(text(display).size(12).font(Font::MONOSPACE));
        r = r.push(Space::with_width(Length::Fill));
        r = r.push(
            button(text("Copy").size(11))
                .on_press(on_copy)
                .style(button::text),
        );
