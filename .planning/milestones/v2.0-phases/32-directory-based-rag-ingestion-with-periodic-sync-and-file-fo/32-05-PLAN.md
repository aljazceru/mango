    ```swift
    import SwiftUI
    import UniformTypeIdentifiers

    struct DirectorySourcePicker: UIViewControllerRepresentable {
        let onPicked: (URL, Data) -> Void   // (resolved URL, bookmark data BLOB)
        let onCancel: () -> Void

        func makeUIViewController(context: Context) -> UIDocumentPickerViewController {
            let picker = UIDocumentPickerViewController(forOpeningContentTypes: [.folder], asCopy: false)
            picker.delegate = context.coordinator
            picker.allowsMultipleSelection = false
            return picker
        }

        func updateUIViewController(_ uiViewController: UIDocumentPickerViewController, context: Context) {}
        func makeCoordinator() -> Coordinator { Coordinator(self) }

        final class Coordinator: NSObject, UIDocumentPickerDelegate {
            let parent: DirectorySourcePicker
            init(_ parent: DirectorySourcePicker) { self.parent = parent }

            func documentPicker(_ controller: UIDocumentPickerViewController, didPickDocumentsAt urls: [URL]) {
                guard let url = urls.first else { parent.onCancel(); return }
                guard url.startAccessingSecurityScopedResource() else { parent.onCancel(); return }
                defer { url.stopAccessingSecurityScopedResource() }
                do {
                    let data = try url.bookmarkData(options: .minimalBookmark,
                                                     includingResourceValuesForKeys: nil,
                                                     relativeTo: nil)
                    parent.onPicked(url, data)
                } catch {
                    parent.onCancel()
                }
            }

            func documentPickerWasCancelled(_ controller: UIDocumentPickerViewController) {
                parent.onCancel()
            }
        }
    }

    // MARK: Bookmark lifecycle helpers

    enum BookmarkResolveResult {
        case ok(url: URL, isStale: Bool, refreshedBookmark: Data?)
        case failure
    }

    func resolveBookmark(_ data: Data) -> BookmarkResolveResult {
        var isStale = false
        do {
            let url = try URL(resolvingBookmarkData: data,
                              options: [],
                              relativeTo: nil,
                              bookmarkDataIsStale: &isStale)
            var refreshed: Data? = nil
            if isStale {
                refreshed = try? url.bookmarkData(options: .minimalBookmark,
                                                   includingResourceValuesForKeys: nil,
                                                   relativeTo: nil)
            }
            return .ok(url: url, isStale: isStale, refreshedBookmark: refreshed)
        } catch {
            return .failure
        }
    }

    /// Enumerate folder with exclusion globs + iCloud skip. Returns (relative_path, mtime_secs, size, iCloudSkipped).
    func enumerateDirectory(rootURL: URL, exclusionGlobs: [String]) -> (entries: [(String, Int64, Int64)], skippedCloud: [String], errors: [String]) {
        guard rootURL.startAccessingSecurityScopedResource() else {
            return ([], [], ["startAccessingSecurityScopedResource failed"])
        }
        defer { rootURL.stopAccessingSecurityScopedResource() }

        let fm = FileManager.default
        let keys: [URLResourceKey] = [
            .contentModificationDateKey, .fileSizeKey, .isDirectoryKey,
            .isUbiquitousItemKey, .ubiquitousItemDownloadingStatusKey
        ]
        guard let enumerator = fm.enumerator(at: rootURL, includingPropertiesForKeys: keys, options: []) else {
            return ([], [], ["enumerator unavailable"])
        }

        var entries: [(String, Int64, Int64)] = []
        var skipped: [String] = []
        let excludeMatcher = GlobMatcher(patterns: exclusionGlobs)

        for case let url as URL in enumerator {
            let relPath = url.path.replacingOccurrences(of: rootURL.path + "/", with: "")
            if excludeMatcher.matches(relPath) { continue }
            let values = try? url.resourceValues(forKeys: Set(keys))
            if values?.isDirectory == true { continue }
            // iCloud placeholder check (D-17, Pitfall 3)
            if values?.isUbiquitousItem == true,
               values?.ubiquitousItemDownloadingStatus == .notDownloaded {
                skipped.append(relPath)
                continue
            }
            let mtime = Int64((values?.contentModificationDate?.timeIntervalSince1970 ?? 0))
            let size = Int64(values?.fileSize ?? 0)
            entries.append((relPath, mtime, size))
        }
        return (entries, skipped, [])
    }
    ```

    Add a minimal `GlobMatcher` struct that translates simple gitignore-style patterns to NSPredicate / custom matcher. Matching logic: if pattern ends with `/`, match any path where `relPath.hasPrefix(pattern)`; if pattern starts with `*.`, match extension; else literal hasPrefix match. For parity with Rust `globset`, we keep it deliberately simple on-device and let the Rust side re-validate patterns at save time. (If complex globs are common in user input, a future plan can FFI into `mango_core::rag::directory_sync::validate_glob_pattern` for exhaustive match — for now D-29 allows either path and the simple matcher covers the Obsidian default set.)

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User→UIDocumentPicker | User grants access to folder via system picker |
| Bookmark→app | Bookmark BLOB can only resolve to a URL the user previously granted |
| iCloud→app | iCloud placeholder files flagged by ubiquitousItemDownloadingStatus |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-V4 | Access Control | bookmark resolution | mitigate | iOS sandbox scopes bookmark to user-granted folder; all reads wrapped in startAccessingSecurityScopedResource() + defer stop (D-16) — escape is impossible |
| T-32-V4b | Elevation of Privilege | bookmark-to-path construction | mitigate | Never construct arbitrary paths; only enumerate URLs returned by FileManager.enumerator(at: bookmarkResolvedURL) (D-36) |
| T-32-I3 | Information Disclosure | iCloud placeholder reads | mitigate | Check ubiquitousItemDownloadingStatus == .notDownloaded and skip; never call startDownloadingUbiquitousItem which could leak presence to iCloud servers (D-17) |
| T-32-DoS3 | Denial of Service | Large vault single-batch | mitigate | 50-file chunking before dispatch to Rust; Rust flushes vector index per batch (D-25, D-27) |
</threat_model>
