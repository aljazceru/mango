    ```rust
    fn relative_time_label(last: Option<i64>, now_secs: i64) -> String {
        let Some(t) = last else { return "Never".into() };
        let delta = now_secs - t;
        match delta {
            d if d < 0 => "Just now".into(),
            0..=59 => "Just now".into(),
            60..=3599 => format!("{}m ago", delta / 60),
            3600..=86399 => format!("{}h ago", delta / 3600),
            86400..=172799 => "Yesterday".into(),
            _ => format!("{}d ago", delta / 86400),
        }
    }
    ```
