    ```rust
    #[derive(Debug, Clone, PartialEq)]
    pub struct StoredFingerprint {
        pub file_path: String,
        pub mtime_secs: i64,
        pub size_bytes: i64,
    }

    /// Compare current enumeration (path, mtime_secs, size_bytes) against stored fingerprints.
    /// Returns FileDiff partitioned into added/modified/removed.
    /// Unchanged files (same path + mtime + size) are in NONE of the buckets.
    pub fn diff_files(
        stored: &[StoredFingerprint],
        current: &[(String, i64, i64)],
    ) -> FileDiff {
        use std::collections::HashMap;
        let stored_map: HashMap<&str, (i64, i64)> = stored
            .iter()
            .map(|s| (s.file_path.as_str(), (s.mtime_secs, s.size_bytes)))
            .collect();
        let current_map: HashMap<&str, (i64, i64)> = current
            .iter()
            .map(|(p, m, s)| (p.as_str(), (*m, *s)))
            .collect();

        let mut diff = FileDiff::default();
        for (path, (mtime, size)) in &current_map {
            match stored_map.get(path) {
                None => diff.added.push((path.to_string(), *mtime, *size)),
                Some((sm, ss)) if *sm != *mtime || *ss != *size => {
                    diff.modified.push((path.to_string(), *mtime, *size));
                }
                Some(_) => { /* unchanged */ }
            }
        }
        for s in stored {
            if !current_map.contains_key(s.file_path.as_str()) {
                diff.removed.push(s.file_path.clone());
            }
        }
        diff
    }
    ```

    ```rust
    #[cfg(not(any(target_os = "ios", target_os = "android")))]
    pub fn walk_with_exclusions(
        root: &str,
        exclusion_globs: &[String],
    ) -> anyhow::Result<Vec<(String, i64, i64)>> {
        use ignore::{WalkBuilder, overrides::OverrideBuilder};
        use std::time::UNIX_EPOCH;

        let mut ob = OverrideBuilder::new(root);
        for glob in exclusion_globs {
            let pattern = if glob.starts_with('!') { glob.clone() } else { format!("!{}", glob) };
            ob.add(&pattern).map_err(|e| anyhow::anyhow!("invalid exclusion glob '{}': {}", glob, e))?;
        }
        let overrides = ob.build()?;

        let walker = WalkBuilder::new(root)
            .overrides(overrides)
            .hidden(false)
            .git_ignore(false)
            .standard_filters(false)
            .build();

        let mut out = Vec::new();
        for result in walker {
            let entry = match result { Ok(e) => e, Err(_) => continue };
            if !entry.file_type().map_or(false, |t| t.is_file()) { continue; }
            let meta = match entry.metadata() { Ok(m) => m, Err(_) => continue };
            let mtime_secs = meta.modified()
                .ok()
                .and_then(|t| t.duration_since(UNIX_EPOCH).ok())
                .map(|d| d.as_secs() as i64)
                .unwrap_or(0);
            let size = meta.len() as i64;
            let path = entry.path().to_string_lossy().into_owned();
            out.push((path, mtime_secs, size));
        }
        Ok(out)
    }

    #[cfg(not(any(target_os = "ios", target_os = "android")))]
    pub fn validate_exclusion_glob(glob: &str) -> anyhow::Result<()> {
        use ignore::overrides::OverrideBuilder;
        let mut ob = OverrideBuilder::new(".");
        let pattern = if glob.starts_with('!') { glob.to_string() } else { format!("!{}", glob) };
        ob.add(&pattern).map_err(|e| anyhow::anyhow!("{}", e))?;
        ob.build().map_err(|e| anyhow::anyhow!("{}", e))?;
        Ok(())
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User→exclusion_globs | User-supplied glob strings cross into `ignore::OverrideBuilder` |
| User→walk_root | Walk root is user-picked path; never arbitrary from code |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-V5 | Tampering | walk_with_exclusions(exclusion_globs) | mitigate | `OverrideBuilder::add()` returns Err on malformed patterns — surfaced via anyhow to caller; `!../../etc/passwd`-style patterns are scoped to walk root by ignore crate design and cannot escape (verified by `test_walk_path_traversal_inert`) |
| T-32-V5b | Input Validation | validate_glob_pattern | mitigate | `globset::Glob::new` rejects malformed globs before they reach storage; tested in `test_validate_exclusion_glob_malformed` |
| T-32-V4 | Access Control | walk_with_exclusions(root) | accept | Root is user-granted path (Desktop rfd::pick_folder) — walker only reads paths enumerated under that root; no symlink-follow escapes because `WalkBuilder::follow_links` defaults to false |
</threat_model>
