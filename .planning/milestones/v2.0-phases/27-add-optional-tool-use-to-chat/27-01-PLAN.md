    ```rust
    /// Build the tool subset for chat tool use (Phase 27).
    /// Excludes `finish` (not meaningful in chat). Conditionally includes
    /// `search_documents`/`read_document` only when docs are attached.
    /// Excludes `web_search` when no Brave API key is set.
    pub fn build_chat_tools(
        include_doc_search: bool,
        brave_api_key_set: bool,
    ) -> Vec<ChatCompletionTools> {
        let all = build_agent_tools();
        all.into_iter()
            .filter(|tool| {
                let name = match tool {
                    ChatCompletionTools::Function(ref t) => t.function.name.as_str(),
                };
                match name {
                    "finish" => false,
                    "search_documents" | "read_document" => include_doc_search,
                    "web_search" => brave_api_key_set,
                    _ => true,
                }
            })
            .collect()
    }
    ```
